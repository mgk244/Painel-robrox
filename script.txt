local function AddToggle(name, page, key)
    local t = Instance.new("TextButton", page)
    t.Size = UDim2.new(0.95, 0, 0, 38); t.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    t.Text = name..": OFF"; t.TextColor3 = Color3.new(0.6, 0.6, 0.6); t.Font = Enum.Font.GothamBold
    Instance.new("UICorner", t).CornerRadius = UDim.new(0, 8)
    t.MouseButton1Click:Connect(function()
        Settings[key] = not Settings[key]
        t.Text = name..": "..(Settings[key] and "ON" or "OFF")
        t.TextColor3 = Settings[key] and Settings.MainColor or Color3.new(0.6, 0.6, 0.6)
    end)
end

local function AddColorSlider(name, page, colorKey)
    local label = Instance.new("TextLabel", page)
    label.Size = UDim2.new(0.95, 0, 0, 20); label.BackgroundTransparency = 1; label.Text = name; label.TextColor3 = Color3.new(1,1,1); label.Font = Enum.Font.GothamBold; label.TextSize = 12
    
    local slider = Instance.new("TextBox", page)
    slider.Size = UDim2.new(0.95, 0, 0, 30); slider.BackgroundColor3 = Color3.fromRGB(30,30,30); slider.Text = tostring(Settings[colorKey]); slider.TextColor3 = Color3.new(1,1,1)
    Instance.new("UICorner", slider)
    
    slider.FocusLost:Connect(function()
        local val = tonumber(slider.Text) or 0
        Settings[colorKey] = math.clamp(val, 0, 255)
        Settings.MainColor = Color3.fromRGB(Settings.R, Settings.G, Settings.B)
        OpenBtn.TextColor3 = Settings.MainColor
    end)
end

-- ESP CACHE (FIX: FILLED = FALSE)
local espCache = {}
local function makeEsp(plr)
    local b = Drawing.new("Square")
    b.Thickness = 1.5
    b.Filled = false -- AQUI ESTÁ O SEGREDO: FALSE TIRA O VERMELHO DO MEIO
    b.Visible = false
    
    local l = Drawing.new("Line")
    l.Thickness = 1
    l.Visible = false
    
    local h = Drawing.new("Line")
    h.Thickness = 2
    h.Visible = false
    
    espCache[plr] = {Box = b, Line = l, Health = h}
end

Players.PlayerAdded:Connect(makeEsp)
for _, p in pairs(Players:GetPlayers()) do if p ~= LocalPlayer then makeEsp(p) end end

-- LOOP PRINCIPAL
RunService.RenderStepped:Connect(function()
    local fps = math.floor(1/RunService.RenderStepped:Wait())
    StatTxt.Text = "USUÁRIO: "..LocalPlayer.Name.."\nFPS: "..fps.."\nPING: "..Stats.Network.ServerStatsItem["Data Ping"]:GetValueString():split(" ")[1].."ms"
    
    FOVCircle.Visible = Settings.FovVisible
    FOVCircle.Radius = Settings.FovRadius
    FOVCircle.Color = Settings.MainColor
    FOVCircle.Position = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)

    local target = nil
    local distHead = Settings.FovRadius
    
    for plr, obj in pairs(espCache) do
        local char = plr.Character
        local head = char and char:FindFirstChild("Head")
        local hum = char and char:FindFirstChild("Humanoid")

        if head and hum and hum.Health > 0 then
            local pos, vis = Camera:WorldToViewportPoint(head.Position)
            if vis then
                -- Tamanho dinâmico baseado na distância
                local sY = 3500 / pos.Z
                local sX = 2200 / pos.Z
                
                -- Ajuste da Box (Apenas contorno)
                obj.Box.Visible = Settings.EspBox
                obj.Box.Size = Vector2.new(sX, sY)
                obj.Box.Position = Vector2.new(pos.X - sX/2, pos.Y - sY/2)
                obj.Box.Color = Settings.MainColor

                -- Linha (Tracer)
                obj.Line.Visible = Settings.EspLine
                obj.Line.From = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y)
                obj.Line.To = Vector2.new(pos.X, pos.Y + sY/2)
                obj.Line.Color = Settings.MainColor

                -- Barra de Vida
                obj.Health.Visible = Settings.EspHealth
                obj.Health.From = Vector2.new(pos.X - sX/2 - 5, pos.Y + sY/2)
                obj.Health.To = Vector2.new(pos.X - sX/2 - 5, (pos.Y + sY/2) - (sY * (hum.Health/hum.MaxHealth)))
                obj.Health.Color = Color3.new(0,1,0)

                local mag = (Vector2.new(pos.X, pos.Y) - (Camera.ViewportSize/2)).Magnitude
                if mag < distHead then distHead = mag; target = head end
            else obj.Box.Visible = false; obj.Line.Visible = false; obj.Health.Visible = false end
        else obj.Box.Visible = false; obj.Line.Visible = false; obj.Health.Visible = false end
    end

    -- AIMBOT LOGIC
    if target then
        local tCFrame = CFrame.new(Camera.CFrame.Position, target.Position)
        if Settings.Aimlock then Camera.CFrame = tCFrame
        elseif Settings.Aimbot then Camera.CFrame = Camera.CFrame:Lerp(tCFrame, 0.15) end
    end

    -- PHYSICS (SPIN, JUMP, NOCLIP)
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if hrp then
        if Settings.Spinbot then
            hrp.CFrame *= CFrame.Angles(0, math.rad(Settings.SpinSpeed), 0)
            if Settings.YeetForce then
                for _, p in pairs(Players:GetPlayers()) do
                    if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                        local eHrp = p.Character.HumanoidRootPart
                        if (hrp.Position - eHrp.Position).Magnitude < 12 then
                            eHrp.Velocity = Vector3.new(0, 1000, 0) -- Yeet Up
                        end
                    end
                end
            end
        end
        if Settings.InfJump and hrp.Velocity.Y < 0 then
            hrp.Velocity = Vector3.new(hrp.Velocity.X, Settings.FallDelay, hrp.Velocity.Z)
        end
        if Settings.Noclip then
            for _, v in pairs(LocalPlayer.Character:GetDescendants()) do 
                if v:IsA("BasePart") then v.CanCollide = false end 
            end
        end
    end
end)

UIS.JumpRequest:Connect(function()

    end
